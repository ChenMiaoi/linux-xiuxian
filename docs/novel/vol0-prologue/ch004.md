# 第四章：寄存器与指令

> **涉及内核源码：** `arch/x86/include/asm/ptrace.h`, `arch/x86/entry/entry_64.S`

## 楔子

林小源理解了编译器的秘法，但他知道，编译器只是中间人。

真正执行代码的，是 CPU。

而 CPU 的核心，是**寄存器**。

他开始观察 CPU 的行为。他看到数据在寄存器之间流动——`RAX` 存着返回值，`RDI` 存着第一个参数，`RSP` 指向栈顶。每一次函数调用，寄存器都会被保存和恢复；每一次算术运算，结果都会被写入某个寄存器。

他数了数，一共 16 个通用寄存器。每个 64 位宽。

他试着理解它们的用途。`RAX` 是返回值，`RCX` 是计数器，`RBP` 是帧指针，`RSP` 是栈指针……每一个寄存器都有约定俗成的用途，就像每一个修士都有自己的职责。

但他最感兴趣的，是一个叫 `RFLAGS` 的寄存器。

它不存数据，只存状态——进位、零、符号、溢出……以及一个叫 `IF` 的标志位，控制着 CPU 是否响应中断。

他当时不知道，这个小小的标志位，日后将成为他面对的第一个真正的危机。

## 一、x86-64 通用寄存器

寄存器是 CPU 内部的高速存储。x86-64 有 16 个通用寄存器，每个都有特定的用途约定。

```c:playground:寄存器概览 — CPU 的丹田
#include <stdio.h>

/*
 * x86-64 通用寄存器（Linux/AMD64 调用约定）
 *
 * 64 位   32 位   16 位   8 位    用途
 * ─────────────────────────────────────────
 * RAX     EAX     AX      AL      返回值 / 乘除法隐含操作数
 * RBX     EBX     BX      BL      被调用者保存（callee-saved）
 * RCX     ECX     CX      CL      第 4 个参数 / 循环计数
 * RDX     EDX     DX      DL      第 3 个参数 / 乘除法隐含操作数
 * RSI     ESI     SI      SIL     第 2 个参数
 * RDI     EDI     DI      DIL     第 1 个参数
 * RBP     EBP     BP      BPL     帧指针（可选）
 * RSP     ESP     SP      SPL     栈指针（不能直接操作）
 * R8-R15  R8D-R15D R8W-R15W R8B-R15B  R8=第5参数，R9=第6参数
 */

int main() {
    printf("=== x86-64 通用寄存器 ===\n\n");

    printf("寄存器      大小    用途\n");
    printf("─────────────────────────────────────────\n");
    printf("RAX         8B      返回值 / 乘除法\n");
    printf("RBX         8B      被调用者保存\n");
    printf("RCX         8B      第 4 参数 / 计数\n");
    printf("RDX         8B      第 3 参数 / 乘除法\n");
    printf("RSI         8B      第 2 参数\n");
    printf("RDI         8B      第 1 参数\n");
    printf("RBP         8B      帧指针\n");
    printf("RSP         8B      栈指针\n");
    printf("R8          8B      第 5 参数\n");
    printf("R9          8B      第 6 参数\n");
    printf("R10         8B      调用者保存\n");
    printf("R11         8B      调用者保存\n");
    printf("R12-R15     8B      被调用者保存\n");

    printf("\n函数调用约定 (System V AMD64 ABI):\n");
    printf("  参数依次放入: RDI, RSI, RDX, RCX, R8, R9\n");
    printf("  多余的参数压栈\n");
    printf("  返回值放在 RAX\n");

    printf("\n内核中的区别:\n");
    printf("  系统调用号放在 RAX\n");
    printf("  系统调用参数: RDI, RSI, RDX, R10, R8, R9\n");
    printf("  (注意: R10 代替了 RCX，因为 syscall 指令会破坏 RCX)\n");

    return 0;
}
```

## 二、标志寄存器 RFLAGS

```c:playground:RFLAGS — CPU 的状态面板
#include <stdio.h>
#include <stdint.h>

/*
 * RFLAGS 重要标志位:
 *
 * 位   缩写  名称              含义
 * ─────────────────────────────────────────
 * 0    CF    Carry Flag        进位 / 借位
 * 2    PF    Parity Flag       低 8 位奇偶校验
 * 6    ZF    Zero Flag         结果为零
 * 7    SF    Sign Flag         结果为负
 * 11   OF    Overflow Flag     有符号溢出
 * 9    IF    Interrupt Enable  中断使能（内核中极重要）
 * 10   DF    Direction Flag    字符串操作方向
 * 16   RF    Resume Flag       调试恢复
 * 17   VM    Virtual 8086      虚拟 8086 模式
 */

int main() {
    printf("=== RFLAGS 标志位 ===\n\n");

    /* 演示进位标志 CF */
    uint8_t a = 200, b = 100;
    uint8_t sum = a + b;
    printf("200 + 100 (uint8_t): %u (溢出! 实际是 44)\n", sum);
    printf("  CF=1: 发生了进位\n\n");

    /* 演示零标志 ZF */
    int x = 5 - 5;
    printf("5 - 5 = %d\n", x);
    printf("  ZF=1: 结果为零\n\n");

    /* 演示符号标志 SF */
    int y = 3 - 7;
    printf("3 - 7 = %d\n", y);
    printf("  SF=1: 结果为负\n\n");

    printf("内核中最重要的标志:\n");
    printf("  IF (中断使能): cli() 清除, sti() 设置\n");
    printf("  控制 CPU 是否响应外部中断\n");
    printf("  内核在关键代码段会关闭中断\n");

    return 0;
}
```

`IF` 标志是内核的生命线。`cli()` 清除它（关中断），`sti()` 设置它（开中断）。内核在修改关键数据结构时必须关中断，防止中断处理函数看到不一致的状态。

## 三、常用指令

```c:playground:汇编指令速查 — 内核常用
#include <stdio.h>

/*
 * 模拟汇编指令的效果（用 C 伪代码展示）
 */

static long simulate_add(long a, long b) {
    long result;
    /* add rax, rbx */
    result = a + b;
    return result;
}

static long simulate_sub(long a, long b) {
    long result;
    /* sub rax, rbx */
    result = a - b;
    return result;
}

static long simulate_mul(long a, long b) {
    long result;
    /* imul rax, rbx */
    result = a * b;
    return result;
}

static long simulate_div(long a, long b, long *remainder) {
    /* cqo; idiv rbx */
    *remainder = a % b;
    return a / b;
}

static long simulate_shift_left(long val, int count) {
    /* shl rax, cl */
    return val << count;
}

static long simulate_shift_right(long val, int count) {
    /* shr rax, cl (逻辑右移) or sar rax, cl (算术右移) */
    return (long)((unsigned long)val >> count);
}

static long simulate_and(long a, long b) { return a & b; }
static long simulate_or(long a, long b)  { return a | b; }
static long simulate_xor(long a, long b) { return a ^ b; }
static long simulate_not(long a)          { return ~a; }

int main() {
    printf("=== 常用汇编指令 (C 等价) ===\n\n");

    printf("数据传送:\n");
    printf("  mov  rax, rbx    →  rax = rbx\n");
    printf("  lea  rax, [rbx+rcx*4+8]  →  rax = rbx + rcx*4 + 8\n");
    printf("  xchg rax, rbx    →  swap(rax, rbx)\n\n");

    printf("算术运算:\n");
    long a = 100, b = 37;
    printf("  add:  %ld + %ld = %ld\n", a, b, simulate_add(a, b));
    printf("  sub:  %ld - %ld = %ld\n", a, b, simulate_sub(a, b));
    printf("  imul: %ld * %ld = %ld\n", a, b, simulate_mul(a, b));
    long rem;
    printf("  idiv: %ld / %ld = %ld 余 %ld\n", a, b,
           simulate_div(a, b, &rem), rem);
    printf("  inc/dec: 加 1 / 减 1\n");
    printf("  neg:  取反 (0 - val)\n\n");

    printf("位运算:\n");
    printf("  and:  %ld & %ld = %ld\n", a, b, simulate_and(a, b));
    printf("  or:   %ld | %ld = %ld\n", a, b, simulate_or(a, b));
    printf("  xor:  %ld ^ %ld = %ld\n", a, b, simulate_xor(a, b));
    printf("  not:  ~%ld = %ld\n", a, simulate_not(a));
    printf("  shl:  %ld << 3 = %ld\n", a, simulate_shift_left(a, 3));
    printf("  shr:  %ld >> 3 = %ld\n", a, simulate_shift_right(a, 3));

    printf("\n栈操作:\n");
    printf("  push rax    →  RSP -= 8; *RSP = RAX\n");
    printf("  pop  rax    →  RAX = *RSP; RSP += 8\n");
    printf("  call func   →  push RIP; jmp func\n");
    printf("  ret         →  pop RIP\n");

    printf("\n比较与跳转:\n");
    printf("  cmp  rax, rbx  →  设置 RFLAGS (rax - rbx)\n");
    printf("  test rax, rax  →  设置 RFLAGS (rax & rax)\n");
    printf("  je / jne       →  相等 / 不等时跳转\n");
    printf("  jg / jl        →  大于 / 小于时跳转（有符号）\n");
    printf("  ja / jb        →  大于 / 小于时跳转（无符号）\n");
    printf("  jmp            →  无条件跳转\n");

    return 0;
}
```

## 四、栈帧 — 函数调用的真相

```c:playground:栈帧 — 函数调用的真相
#include <stdio.h>

/*
 * 函数调用时栈的变化（简化版）:
 *
 * 高地址
 * ┌──────────────────┐
 * │  调用者的栈帧    │
 * ├──────────────────┤
 * │  返回地址 (RIP)  │ ← call 指令压入
 * ├──────────────────┤
 * │  保存的 RBP      │ ← 函数序言压入
 * ├──────────────────┤ ← RBP 指向这里
 * │  局部变量        │
 * │  ...             │
 * ├──────────────────┤ ← RSP 指向这里
 * 低地址
 */

static void show_addrs(int depth, long *caller_sp) {
    int local = depth;
    long dummy;

    printf("depth %d: local @ %p, 值 = %d\n",
           depth, (void *)&local, local);

    if (depth > 0) {
        show_addrs(depth - 1, &dummy);
    }
}

static void func_b(int x) {
    int local_b = x * 2;
    printf("func_b: local_b @ %p, 值 = %d\n", (void *)&local_b, local_b);
}

static void func_a(int x) {
    int local_a = x + 10;
    printf("func_a: local_a @ %p, 值 = %d\n", (void *)&local_a, local_a);
    func_b(local_a);
}

int main() {
    printf("=== 栈帧 ===\n\n");

    printf("1. 栈向下增长:\n");
    show_addrs(3, NULL);

    printf("\n2. 函数调用链:\n");
    func_a(42);

    printf("\n3. 关键点:\n");
    printf("   - 每次函数调用，RSP 减小（栈向下增长）\n");
    printf("   - 局部变量在栈帧中，地址递减\n");
    printf("   - 内核栈很小（8KB），不能深度递归!\n");
    printf("   - 内核用 objtool 检测栈溢出风险\n");

    return 0;
}
```

## 五、系统调用 — 从用户态到内核态

```c:playground:系统调用 — 跨越特权级
#include <stdio.h>
#include <unistd.h>
#include <sys/syscall.h>

/*
 * 系统调用过程:
 *
 * 用户态                              内核态
 * ──────                              ──────
 * 1. RAX = 系统调用号
 * 2. RDI = 第 1 个参数
 *    RSI = 第 2 个参数
 *    RDX = 第 3 个参数
 *    R10 = 第 4 个参数
 *    R8  = 第 5 个参数
 *    R9  = 第 6 个参数
 * 3. syscall 指令 ──────────────────→ 4. 保存用户态寄存器
 *                                      5. 查找系统调用表
 *                                      6. 执行处理函数
 *                   ←──────────────── 7. 结果放入 RAX
 * 8. 检查 RAX（负值 = 错误）           7. 恢复用户态寄存器
 */

int main() {
    printf("=== 系统调用 ===\n\n");

    /* write(1, "Hello\n", 6) 的底层过程 */
    printf("1. write() 系统调用:\n");
    printf("   RAX = %d (SYS_write)\n", SYS_write);
    printf("   RDI = 1 (stdout)\n");
    printf("   RSI = 缓冲区地址\n");
    printf("   RDX = 6 (字节数)\n");
    printf("   → syscall 指令\n\n");

    /* getpid() */
    pid_t pid = getpid();
    printf("2. getpid() = %d\n", pid);
    printf("   系统调用号: %d (SYS_getpid)\n\n", SYS_getpid);

    /* 展示系统调用号 */
    printf("3. 常用系统调用号 (x86-64):\n");
    printf("   SYS_read       = %d\n", SYS_read);
    printf("   SYS_write      = %d\n", SYS_write);
    printf("   SYS_open       = %d\n", SYS_open);
    printf("   SYS_close      = %d\n", SYS_close);
    printf("   SYS_mmap       = %d\n", SYS_mmap);
    printf("   SYS_ioctl      = %d\n", SYS_ioctl);
    printf("   SYS_exit       = %d\n", SYS_exit);

    printf("\n4. syscall 指令会:\n");
    printf("   - 从 Ring 3 (用户态) 切换到 Ring 0 (内核态)\n");
    printf("   - 保存 RIP 到 RCX, RFLAGS 到 R11\n");
    printf("   - 跳转到 MSR_LSTAR 中存储的内核入口地址\n");
    printf("   - 切换到内核栈\n");

    return 0;
}
```

`syscall` 指令是用户态进入内核的唯一合法通道。它会自动保存用户态的 RIP（到 RCX）和 RFLAGS（到 R11），然后跳转到内核预设的系统调用入口。

## 六、特权级 — Ring 0 到 Ring 3

```c
/*
 * x86 特权级:
 *
 * Ring 0 — 内核态 (Kernel)
 *   - 可以执行所有指令
 *   - 可以访问所有内存
 *   - 可以修改 CR3（页表基址）
 *   - 可以执行 I/O 指令
 *
 * Ring 1, 2 — 通常不使用
 *
 * Ring 3 — 用户态 (User)
 *   - 不能执行特权指令
 *   - 不能访问内核内存
 *   - 必须通过 syscall 请求内核服务
 *
 * CPL (Current Privilege Level) 存在 CS 寄存器的低 2 位:
 *   00 = Ring 0
 *   11 = Ring 3
 */
```

### 道藏笔记

::: tip 内核启示
**汇编是内核的母语。** 虽然内核主要用 C 编写，但启动代码、上下文切换、系统调用入口、中断处理——这些最核心的部分都是汇编。

在内核源码中，`arch/x86/entry/entry_64.S` 是系统调用的入口，`arch/x86/kernel/process.c` 中的 `__switch_to()` 涉及上下文切换的汇编。不理解汇编，就无法理解这些关键路径。

值得注意的是，`syscall` 指令是用户态进入内核的**唯一合法通道**。它将 CPU 从 Ring 3 切换到 Ring 0，跳转到内核预设的入口地址。这个入口地址存储在 MSR 寄存器 `LSTAR` 中，由内核在启动时设置。

但"唯一合法通道"这个说法，日后会变得意味深长。如果有人能修改 `sys_call_table`——那个存储所有系统调用处理函数地址的表——那么每一个用户态的请求都会经过他的手。内核中称这种技术为"系统调用表 hook"，它是 rootkit 最常用的手段之一。

而 `RFLAGS` 中的 `IF` 标志（中断使能），控制着 CPU 是否响应外部中断。关中断意味着整个世界暂停运转——这是保护临界区的最后手段，也是最危险的手段。内核栈只有 8KB，一次深度递归就可能让它溢出。这些看似抽象的概念，日后都会以最具体的方式呈现在你面前。
:::
