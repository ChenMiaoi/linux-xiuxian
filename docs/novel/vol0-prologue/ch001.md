# 第一章：指针之道

> **涉及内核源码：** `include/linux/types.h`, `arch/x86/include/asm/page_types.h`

## 一、指针的本质

指针不是魔法。它只是一个存着内存地址的变量。

在 64 位系统上，一个指针占 8 字节，恰好能装下一个 64 位的地址。就这么简单。

```c:playground:指针与地址 — 万物皆有居所
#include <stdio.h>
#include <stdint.h>

int main() {
    int x = 42;
    int *p = &x;

    printf("=== 指针与地址 ===\n");
    printf("x 的值:       %d\n", x);
    printf("x 的地址:     %p\n", (void *)&x);
    printf("p 的值:       %p\n", (void *)p);
    printf("*p 的值:      %d\n", *p);
    printf("p 自身的地址: %p\n", (void *)&p);
    printf("\n");
    printf("指针大小:     %zu 字节\n", sizeof(int *));
    printf("int 大小:     %zu 字节\n", sizeof(int));

    return 0;
}
```

内核中的指针和用户态的指针没有任何区别。区别在于：内核的指针指向的是真实的物理内存、硬件寄存器、内核数据结构——错一步，系统崩溃。

## 二、指针的算术

指针加 1，实际增加的字节数取决于它指向的类型。

```c:playground:指针算术 — 步伐由类型决定
#include <stdio.h>
#include <stdint.h>

int main() {
    int arr[5] = {10, 20, 30, 40, 50};
    int *p = arr;

    printf("=== 指针算术 ===\n");
    for (int i = 0; i < 5; i++) {
        printf("arr[%d] = %d, 地址: %p, 偏移: %td 字节\n",
               i, *(p + i), (void *)(p + i),
               (char *)(p + i) - (char *)arr);
    }

    printf("\n=== 不同类型的步长 ===\n");
    printf("int*   +1: +%zu 字节\n", sizeof(int));
    printf("char*  +1: +%zu 字节\n", sizeof(char));
    printf("long*  +1: +%zu 字节\n", sizeof(long));

    return 0;
}
```

内核中大量使用指针算术来遍历数据结构。理解 `p + 1` 跳过多少字节，是读懂内核代码的第一步。

## 三、void 指针 — 万能容器

`void *` 是内核中最常见的指针类型。它没有类型信息，可以指向任何东西，但不能直接解引用——必须先转型。

```c:playground:void 指针 — 无相之体
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct task_struct {
    long state;
    int pid;
    char comm[16];
};

int main() {
    struct task_struct task = { .state = 0, .pid = 42, .comm = "demo" };
    void *vp = &task;

    /* void* 可以赋值给任何指针类型 */
    struct task_struct *tp = (struct task_struct *)vp;

    printf("=== void* 转型 ===\n");
    printf("原始 PID: %d\n", task.pid);
    printf("转型后:   %d\n", tp->pid);
    printf("名称:     %s\n", tp->comm);
    printf("地址相同: %s\n", vp == tp ? "是" : "否");

    return 0;
}
```

内核的 `kmalloc()` 返回 `void *`，`copy_to_user()` 接受 `void *`。`void *` 是内核的通用货币。

## 四、函数指针 — 内核的多态

函数指针让 C 语言拥有了面向对象的多态能力。内核的 `file_operations`、`vm_operations_struct` 等结构体，本质上就是虚函数表。

```c:playground:函数指针 — 虚函数表的真相
#include <stdio.h>

/* 模拟内核的 file_operations */
struct file_operations {
    int (*open)(const char *name);
    int (*read)(char *buf, int size);
    int (*write)(const char *buf, int size);
    int (*close)(void);
};

static int my_open(const char *name) {
    printf("[open] 打开: %s\n", name);
    return 0;
}

static int my_read(char *buf, int size) {
    const char *data = "Hello from kernel!";
    int len = 0;
    while (data[len] && len < size - 1) {
        buf[len] = data[len];
        len++;
    }
    buf[len] = '\0';
    printf("[read] 读取: %s\n", buf);
    return len;
}

static int my_write(const char *buf, int size) {
    printf("[write] 写入: %.*s\n", size, buf);
    return size;
}

static int my_close(void) {
    printf("[close] 关闭\n");
    return 0;
}

int main() {
    struct file_operations fops = {
        .open  = my_open,
        .read  = my_read,
        .write = my_write,
        .close = my_close,
    };

    char buf[64];

    printf("=== 函数指针：模拟 file_operations ===\n");
    fops.open("demo.txt");
    fops.read(buf, sizeof(buf));
    fops.write("test data", 9);
    fops.close();

    printf("\n函数指针大小: %zu 字节\n", sizeof(fops.open));

    return 0;
}
```

当你在内核中看到 `file->f_op->read(...)` 时，这就是多态——不同文件系统提供不同的 `read` 实现，通过函数指针动态分发。

## 五、栈与堆 — 内存的两重天

```c:playground:栈与堆 — 内存的阴阳
#include <stdio.h>
#include <stdlib.h>

int main() {
    /* 栈上分配：自动管理，函数返回即释放 */
    int stack_var = 100;

    /* 堆上分配：手动管理，必须显式释放 */
    int *heap_var = malloc(sizeof(int));
    *heap_var = 200;

    printf("=== 栈与堆 ===\n");
    printf("栈变量: 值=%d, 地址=%p\n", stack_var, (void *)&stack_var);
    printf("堆变量: 值=%d, 地址=%p\n", *heap_var, (void *)heap_var);
    printf("\n");
    printf("栈地址通常在高处，堆地址通常在低处\n");
    printf("栈向下增长，堆向上增长\n");

    /* 内核中的栈非常小（通常 8KB 或 16KB） */
    printf("\n内核栈大小: 通常 8KB-16KB（不能递归！）\n");

    free(heap_var);
    return 0;
}
```

内核栈极其宝贵——通常只有 8KB（x86）或 16KB（x86_64）。在栈上分配大数组、深度递归，都会导致栈溢出。内核开发者必须时刻警惕栈的使用。

## 六、const 和 restrict

```c
/* const：承诺不修改 */
void read_data(const char *buf);       /* 不会修改 buf */
void set_value(int * const ptr);       /* 不能修改指针本身 */

/* restrict：承诺独占访问（编译器优化用） */
void copy(int * restrict dst, const int * restrict src, int n);
```

`const` 在内核中随处可见，尤其是参数声明。`restrict` 告诉编译器：这个指针是访问这块内存的唯一途径，可以放心优化。

## 七、内核中的类型

内核不使用标准的 `int`、`long`——这些类型在不同架构上大小不同。内核定义了自己的类型：

```c
/* include/linux/types.h */
typedef unsigned char      u8;
typedef unsigned short     u16;
typedef unsigned int       u32;
typedef unsigned long long u64;

typedef signed char        s8;
typedef signed short       s16;
typedef signed int         s32;
typedef signed long long   s64;

/* 指针大小的整数 */
typedef unsigned long      uintptr_t;
typedef signed long        intptr_t;

/* 布尔值 */
typedef _Bool              bool;
#define true  1
#define false 0
```

在内核代码中，你会看到 `u32` 而不是 `unsigned int`，看到 `s64` 而不是 `long long`。这不是偏好，是纪律——类型大小必须明确，不能依赖编译器的解释。

### 道藏笔记

::: tip 内核启示
**指针是内核的语言。** 内核中的一切——进程、文件、网络包、设备——最终都是通过指针连接的。`task_struct` 之间用指针串成链表，文件对象用指针关联到 inode，页表本身就是多级指针树。

理解指针，就是理解内核的思维方式。
:::
