# 第五章：内联汇编

> **涉及内核源码：** `arch/x86/include/asm/processor.h`, `include/linux/percpu.h`

## 楔子

林小源理解了寄存器，但他发现了一个矛盾。

内核是用 C 写的，但有些操作——比如读取控制寄存器、发起系统调用、执行原子操作——C 语言无法直接完成。它们需要特定的汇编指令。

但内核不能全是汇编。汇编难写、难读、难维护。

怎么办？

答案是**内联汇编**——在 C 代码中嵌入汇编指令。

他第一次看到 `asm volatile("mov %%cr3, %0" : "=r"(val))` 这行代码时，完全不知道它在说什么。但当他理解了约束（`"=r"`、`"+m"`、`"i"`）的含义后，他意识到这是一种极其精妙的设计——C 编译器负责寄存器分配和代码布局，程序员负责指定关键的汇编指令。

两者配合，既高效又安全。

但"安全"这个词，只在你写对了的时候成立。

## 一、基本语法

GCC 的内联汇编允许在 C 代码中直接嵌入汇编指令。

```c
asm("汇编指令");
asm volatile("汇编指令");  /* volatile: 防止编译器优化掉
```

`volatile` 告诉编译器：这条汇编有副作用，不能省略，不能移动顺序。内核中的汇编几乎都用 `volatile`。

## 二、带约束的内联汇编

```c:playground:内联汇编 — 约束的语法
#include <stdio.h>

/*
 * 完整语法:
 * asm volatile(
 *     "汇编模板"
 *     : 输出操作数    (可选)
 *     : 输入操作数    (可选)
 *     : 破坏列表      (可选)
 * );
 *
 * 约束字符 (x86-64):
 *   "r"  — 通用寄存器
 *   "a"  — RAX
 *   "b"  — RBX
 *   "c"  — RCX
 *   "d"  — RDX
 *   "S"  — RSI
 *   "D"  — RDI
 *   "i"  — 立即数
 *   "m"  — 内存
 *   "0"  — 与第 0 个操作数相同的约束
 */

int main() {
    printf("=== 内联汇编基础 ===\n\n");

    /* 1. 最简单的汇编 */
    int x;
    asm volatile("mov $42, %0" : "=r"(x));
    printf("mov $42 → x = %d\n\n", x);

    /* 2. 输入输出 */
    int a = 10, b = 20, result;
    asm volatile(
        "addl %1, %2\n\t"
        "movl %2, %0"
        : "=r"(result)
        : "r"(a), "r"(b)
    );
    printf("addl: %d + %d = %d\n\n", a, b, result);

    /* 3. 使用特定寄存器 */
    int val = 100;
    asm volatile(
        "inc %%eax"
        : "+a"(val)    /* + 表示既是输入又是输出 */
    );
    printf("inc: 100 → %d\n\n", val);

    /* 4. 读取 RFLAGS */
    unsigned long flags;
    asm volatile("pushfq; popq %0" : "=r"(flags));
    printf("RFLAGS = 0x%lx\n", flags);
    printf("  IF (中断使能): %s\n", (flags & 0x200) ? "是" : "否");

    return 0;
}
```

## 三、内核中的常用内联汇编

```c:playground:内核内联汇编实战
#include <stdio.h>
#include <stdint.h>

/* 读取 CR3（页表基址寄存器）— 内核中真实存在的函数 */
static inline unsigned long read_cr3(void) {
    unsigned long val;
    asm volatile("mov %%cr3, %0" : "=r"(val));
    return val;
}

/* 读取时间戳计数器 TSC */
static inline uint64_t rdtsc(void) {
    uint32_t lo, hi;
    asm volatile("rdtsc" : "=a"(lo), "=d"(hi));
    return ((uint64_t)hi << 32) | lo;
}

/* CPUID 指令 */
static inline void cpuid(uint32_t leaf, uint32_t *eax, uint32_t *ebx,
                          uint32_t *ecx, uint32_t *edx) {
    asm volatile("cpuid"
        : "=a"(*eax), "=b"(*ebx), "=c"(*ecx), "=d"(*edx)
        : "a"(leaf));
}

/* 内存屏障 */
static inline void mb(void) {
    asm volatile("mfence" ::: "memory");
}

/* 禁用/启用中断（需要 Ring 0 权限） */
static inline void local_irq_disable_sim(void) {
    /* 在用户态会 SIGSEGV，仅展示语法 */
    asm volatile("cli" ::: "memory");
}

int main() {
    printf("=== 内核内联汇编实战 ===\n\n");

    /* TSC */
    uint64_t t1 = rdtsc();
    printf("TSC = %llu\n", (unsigned long long)t1);

    /* 做一些工作 */
    volatile int sum = 0;
    for (int i = 0; i < 1000000; i++) sum += i;

    uint64_t t2 = rdtsc();
    printf("循环后 TSC = %llu\n", (unsigned long long)t2);
    printf("消耗了 %llu 个时钟周期\n\n", (unsigned long long)(t2 - t1));

    /* CPUID */
    uint32_t eax, ebx, ecx, edx;
    cpuid(0, &eax, &ebx, &ecx, &edx);
    printf("CPUID(0):\n");
    printf("  最大叶号: %u\n", eax);
    printf("  厂商字符串: %.4s%.4s%.4s\n",
           (char *)&ebx, (char *)&edx, (char *)&ecx);

    /* 内存屏障说明 */
    printf("\n内存屏障指令:\n");
    printf("  mfence — 全部屏障（读+写）\n");
    printf("  lfence — 读屏障\n");
    printf("  sfence — 写屏障\n");
    printf("  屏障确保之前的内存操作完成后，才执行之后的操作\n");

    return 0;
}
```

## 四、带内存操作数的汇编

```c:playground:内存操作数 — 直接操作内存
#include <stdio.h>
#include <stdint.h>

/* 原子比较并交换 (CAS) */
static inline int atomic_cmpxchg(int *ptr, int old_val, int new_val) {
    int result;
    asm volatile(
        "lock cmpxchgl %2, %1"
        : "=a"(result), "+m"(*ptr)
        : "r"(new_val), "0"(old_val)
        : "memory"
    );
    return result;
}

/* 原子加 */
static inline void atomic_add(int *ptr, int val) {
    asm volatile(
        "lock addl %1, %0"
        : "+m"(*ptr)
        : "ir"(val)
        : "memory"
    );
}

/* 原子交换 */
static inline int xchg(int *ptr, int new_val) {
    int result;
    asm volatile(
        "xchgl %0, %1"
        : "=r"(result), "+m"(*ptr)
        : "0"(new_val)
        : "memory"
    );
    return result;
}

int main() {
    printf("=== 原子操作 ===\n\n");

    /* CAS */
    int val = 100;
    printf("CAS(&val, 100, 200):\n");
    int old = atomic_cmpxchg(&val, 100, 200);
    printf("  返回旧值: %d, val 现在: %d\n\n", old, val);

    /* CAS 失败的情况 */
    printf("CAS(&val, 100, 300) — 期望不匹配:\n");
    old = atomic_cmpxchg(&val, 100, 300);
    printf("  返回旧值: %d (不等于 100，CAS 失败)\n", old);
    printf("  val 仍然是: %d\n\n", val);

    /* 原子加 */
    val = 0;
    printf("原子加:\n");
    for (int i = 0; i < 10; i++) {
        atomic_add(&val, 1);
    }
    printf("  10 次 atomic_add(&val, 1) → val = %d\n\n", val);

    /* 原子交换 */
    val = 42;
    int old_val = xchg(&val, 99);
    printf("xchg: 旧值 = %d, 新值 = %d\n", old_val, val);

    return 0;
}
```

`lock` 前缀是原子操作的关键。它会锁定缓存行（在现代 CPU 上），确保操作的原子性。`cmpxchg` 是实现自旋锁、无锁数据结构的基础。

## 五、系统调用的汇编实现

```c:playground:系统调用 — 手动发起
#include <stdio.h>
#include <unistd.h>
#include <sys/syscall.h>

/*
 * 手动用内联汇编发起 write 系统调用
 * (在 Linux x86-64 上)
 */
static long my_write(int fd, const void *buf, size_t count) {
    long ret;
    asm volatile(
        "syscall"
        : "=a"(ret)
        : "a"(SYS_write), "D"(fd), "S"(buf), "d"(count)
        : "rcx", "r11", "memory"
    );
    return ret;
}

static long my_getpid(void) {
    long ret;
    asm volatile(
        "syscall"
        : "=a"(ret)
        : "a"(SYS_getpid)
        : "rcx", "r11"
    );
    return ret;
}

int main() {
    printf("=== 手动系统调用 ===\n\n");

    /* 用汇编直接调用 write */
    const char *msg = "Hello from inline assembly!\n";
    long len = my_write(1, msg, 29);
    printf("write 返回: %ld\n\n", len);

    /* getpid */
    long pid = my_getpid();
    printf("getpid (汇编): %ld\n", pid);
    printf("getpid (libc): %d\n", getpid());

    printf("\nsyscall 指令的约束:\n");
    printf("  输出: \"=a\"(ret) — 返回值在 RAX\n");
    printf("  输入: \"a\"(调用号), \"D\"(arg1), \"S\"(arg2), \"d\"(arg3)\n");
    printf("  破坏: rcx (保存用户 RIP), r11 (保存用户 RFLAGS)\n");

    return 0;
}
```

### 道藏笔记

::: tip 内核启示
**`syscall` 指令是内核的门户。** 每次用户态程序调用 `read()`、`write()`、`open()`，最终都是通过 `syscall` 指令进入内核。

内核的系统调用入口在 `arch/x86/entry/entry_64.S` 中定义。它保存所有用户态寄存器，查找系统调用表（`sys_call_table`），跳转到对应的处理函数，最后通过 `sysretq` 返回用户态。

原子操作 `lock cmpxchg` 和 `xchg` 在内核中的重要性不言而喻。自旋锁的实现、无锁数据结构、引用计数——这些内核最核心的同步机制，都建立在这些原子指令之上。`lock` 前缀会锁定缓存行，确保操作的原子性。在多核系统上，这是唯一能保证跨 CPU 一致性的手段。

而当你看到 `syscall` 指令自动将 `RIP` 保存到 `RCX`、将 `RFLAGS` 保存到 `R11` 时，你就理解了为什么系统调用的参数传递要用 `R10` 而不是 `RCX`——因为 `syscall` 已经破坏了 `RCX`。这种硬件层面的约束，决定了内核的整个系统调用接口设计。
:::
